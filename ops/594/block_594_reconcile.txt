def iso_epoch(t, naive_utc=True):
    if t is None or str(t).strip() == "":
        return None, None, "no time"
    raw = str(t).strip()
    if DAY_RE.match(raw):
        return D.datetime.fromisoformat(raw + "T00:00:00+00:00").timestamp(), raw, "a date: read as its 00:00Z"
    m = ISO_RE.match(raw)
    if not m:
        return None, raw, "unparsed"
    day, hh, mi, ss, frac, off = m.groups()
    txt = day + "T" + hh + ":" + mi + ":" + ss + ("." + (frac + "000000")[:6] if frac else "")
    if off in ("Z", "z"):
        txt, note = txt + "+00:00", "Z"
    elif off:
        o = off.replace(":", "")
        txt, note = txt + o[:3] + ":" + o[3:], "offset " + off
    elif naive_utc:
        note = "no offset (naive UTC, as the app and the DB write it)"
    else:
        return None, raw, "no reliable time (no Z or offset in the text)"
    try:
        dt = D.datetime.fromisoformat(txt)
    except ValueError:
        return None, raw, "unparsed"
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=D.timezone.utc)
    return dt.timestamp(), raw, note

def txt_epoch(t):
    if t is None or DAY_RE.match(str(t).strip()):
        return None
    e = iso_epoch(t)[0]
    if e is not None:
        return e
    s = str(t).strip().replace("T", " ").rstrip("Z")
    try:
        e = D.datetime.strptime(s[:19], "%Y-%m-%d %H:%M:%S").replace(tzinfo=D.timezone.utc).timestamp()
    except ValueError:
        return None
    if len(s) > 20 and s[19] == "." and s[20:].isdigit():
        e += float("0." + s[20:])
    return e

def parse_env_file(path):
    out = {}
    try:
        text = Path(path).read_text()
    except (OSError, UnicodeDecodeError):
        return out
    for line in text.splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        k, v = line.split("=", 1)
        k = k.strip()
        if k.startswith("export "):
            k = k[7:].strip()
        v = v.strip()
        if len(v) >= 2 and v[0] == v[-1] and v[0] in ("\x22", "\x27"):
            v = v[1:-1]
        out[k] = v
    return out

def is_wal(p):
    try:
        with open(str(p), "rb") as f:
            h = f.read(100)
    except OSError:
        return None
    return len(h) >= 20 and h[:16] == b"SQLite format 3\x00" and h[18] == 2

def signature():
    p = db_path()[0]
    if p is None:
        return {"db": None}
    out = {"db": str(p)}
    for suf in ("", "-wal", "-shm", "-journal"):
        try:
            st = os.stat(str(p) + suf)
            out[p.name + suf] = [st.st_size] if suf == "-shm" else [st.st_size, round(st.st_mtime, 3)]
        except OSError:
            out[p.name + suf] = None
    try:
        out["dir_entries"] = sorted(os.listdir(str(p.parent)))
    except OSError:
        out["dir_entries"] = None
    return out

                      "or run it while an app process holds the DB open (a clean close of the app removes both sidecars)")
    uri = p.resolve().as_uri() + "?mode=ro"
    con = None
    try:
        con = sqlite3.connect(uri, uri=True, timeout=60)
        con.execute("PRAGMA query_only = ON")
        con.execute("SELECT count(*) FROM sqlite_master").fetchone()
    except sqlite3.Error as e:
        if con is not None:
            con.close()
        why = "unreadable read-only (" + uri + "): " + type(e).__name__ + ": " + str(e)
        if wal and miss:

        out["kalshi"] = {k: cl["kalshi"].get(k) for k in ("bid", "ask", "mid", "spread_c")}

def clv_of(pred, cl):
    top = CR._top(pred)
    new, price, book = None, None, None
    if top and CL.priced(cl) and top in cl["fair"]:
        p = {"HOME": pred.home_win_prob, "DRAW": pred.draw_prob, "AWAY": pred.away_win_prob}[top]
        new = p - cl["fair"][top]
        best = cl["best"].get(top)
        if best:
            book, price = best
        elif cl.get("source") == "kalshi_only" and cl["fair"][top] > 0:
            book, price = "kalshi_only", round(1 / cl["fair"][top], 4)
    return top, new, price, book

        head, _, s = line[1:].partition("\x02")
        f = head.split()
        if len(f) < 2 or not f[0].isdigit():
            continue
        subj[f[1]] = s
        if s.startswith(SQUASH_SUBJ):
            rec.add(f[1])
        if len(f) == 2:
            roots.append((int(f[0]), f[1]))
    if roots:
        first = min(roots)[1]
        rec |= {h for _, h in roots if h != first}
    sp = git_run(root, "rev-parse", "--git-path", "shallow")
    if sp and sp.strip():
        q = sp.strip() if os.path.isabs(sp.strip()) else os.path.join(root, sp.strip())
        try:
            rec |= {x for x in open(q).read().split() if x}
        except OSError:
            pass
    RI[root] = {"rec": rec, "subj": subj, "roots": len(roots)}
    return RI[root]

def first_writes(root, spec):
    key = (root, spec)
    if key in FW:
        return FW[key]
    rec = rec_info(root)["rec"]
    out, early, ct, sha = {}, {}, None, None
    for line in (git_run(root, "log", "--all", "--reflog", "--no-renames", "--raw", "--no-abbrev", "--format=%x01%ct %H", "--", *list(spec or ())) or "").splitlines():
        if line.startswith("\x01"):
            f = line[1:].split()
            ct, sha = (int(f[0]), f[1]) if (len(f) == 2 and f[0].isdigit()) else (None, None)
        elif line.startswith(":") and "\t" in line and ct is not None:
            meta, path = line.split("\t", 1)
            f = meta.split()
            if len(f) >= 5 and f[4][:1] in ("A", "M", "T"):
                k = (path, f[3])
                tgt = early if sha in rec else out
                if k not in tgt or ct < tgt[k][0]:
                    tgt[k] = (ct, sha)
    FW[key] = (out, early)
    return FW[key]

def doc_time(p):
    if not str(p).endswith(".json"):
        return None
    try:
        d = json.loads(rd(p))
    except (OSError, ValueError):
        return None
    if not isinstance(d, dict):
        return None
    for k, v in (("exported_at", d.get("exported_at")), ("desk_meta.as_of", g(d, "desk_meta", "as_of"))):
        e, raw, _ = iso_epoch(v)
        if e is not None:
            return e, k, raw
    return None

def git_info(root, spec):
    key = (root, spec)
    if key in GD:
        return GD[key]
    info = {"ok": False, "why": "not a git clone"}
    try:
        top = git_run(root, "rev-parse", "--show-toplevel")
        if top is not None and os.path.realpath(top.strip()) == root:
            ps = ["--"] + list(spec or ())
            tree, last, ct, sha = {}, {}, None, None
            for ent in (git_run(root, "ls-tree", "-r", "-z", "HEAD", *ps) or "").split("\x00"):
                if "\t" in ent:
                    meta, path = ent.split("\t", 1)
                    f = meta.split()
                    if len(f) >= 3 and f[1] == "blob":
                        tree[path] = f[2]
            for line in (git_run(root, "log", "--topo-order", "--no-renames", "--format=%x01%ct %H", "--name-only", "HEAD", *ps) or "").splitlines():
                if line.startswith("\x01"):
                    f = line[1:].split()
                    ct, sha = (int(f[0]), f[1]) if (len(f) == 2 and f[0].isdigit()) else (None, None)
                elif line and ct is not None and line not in last:
                    last[line] = (ct, sha)
            ri = rec_info(root)
            info = {"ok": True, "tree": tree, "last": last, "rec": ri["rec"], "subj": ri["subj"]}
    except Exception as e:
        info = {"ok": False, "why": type(e).__name__ + ": " + str(e)}
    GD[key] = info
    return info

def blob_id(b, n):
    h = hashlib.sha256() if n == 64 else hashlib.sha1()
    h.update(b"blob " + str(len(b)).encode() + b"\x00")
    h.update(b)
    return h.hexdigest()

DT = {}

def dated(p):
    st = os.stat(p)
    k = (os.path.realpath(p), st.st_mtime, st.st_size)
    if k not in DT:
        DT[k] = dated_(p)
    return DT[k]

def dated_(p):
    mt = os.path.getmtime(p)
    rp = os.path.realpath(p)
    for root, spec, label in GIT_ROOTS:
        if not rp.startswith(root + os.sep):
            continue
        rel = os.path.relpath(rp, root).replace(os.sep, "/")
        if spec and not any(rel == s or rel.startswith(s + "/") for s in spec):
            return mt, None
        info = git_info(root, spec)
        if not info["ok"]:
            return mt, "dated by mtime (" + label + ": git unreadable, " + info["why"] + ")"
        want = info["tree"].get(rel)
        if want is None:
            return mt, "dated by mtime (not in " + label + "\x27s HEAD)"
        try:
            same = blob_id(rd(p), len(want)) == want
        except OSError:
            same = False
        if not same:
            return mt, "dated by mtime (differs from " + label + "\x27s HEAD)"
        ct, sha = info["last"].get(rel) or (None, None)
        if ct is None:
            return mt, "dated by mtime (no commit found in " + label + ")"
        mtn = "mtime " + utc(mt) + ", which a checkout or pull can reset"
        if sha in info["rec"]:
            sq = ("re-committed by the weekly squash " if info["subj"].get(sha, "").startswith(SQUASH_SUBJ) else "re-committed by a root commit (orphan or shallow boundary) ") + sha[:10] + " " + utc(ct) + " in " + label
            try:
                fwm, erm = first_writes(root, spec)
                fw, er = fwm.get((rel, want)), erm.get((rel, want))
            except Exception as e:
                fw, er, sq = None, None, sq + " (the --all --reflog read failed: " + type(e).__name__ + ")"
            ub = min(ct, er[0]) if er is not None else ct
            if fw is not None and fw[0] <= ct:
                return fw[0], "dated by its first commit of these bytes " + utc(fw[0]) + " (" + fw[1][:10] + ", reachable through --all --reflog; " + sq + "; " + mtn + ")"
            dt = doc_time(p)
            if dt is not None and dt[0] <= ub:
                return dt[0], "dated by the document\x27s own " + dt[1] + " " + dt[2] + " = " + utc(dt[0]) + " (" + sq + "; no earlier commit of these bytes reachable; " + mtn + ")"
            return UB(ub), ("undatable (squashed): " + sq + "; no earlier commit of these bytes is reachable through --all --reflog and the file carries no exported_at or desk_meta.as_of"
                            " at or before it, so its date is unknown, at or before " + utc(ub) + " (the earliest squash or root commit holding these bytes): listed in every pass whose window starts at or before that (both passes for a squash after the cutoff), "
                            "never silently inside the window (" + mtn + ")")
        return ct, "dated by its last commit " + utc(ct) + " in " + label + " (" + mtn + ("; the mtime is " + BEFORE_LABEL if lab_epoch(mt) and not lab_epoch(ct) else "") + ")"
    return mt, None

def listing(patterns, exclude=None):
    seen, out = set(), []
    for pat in patterns:
        for q in sorted(glob.glob(os.path.expanduser(pat), recursive=True)):
            rp = os.path.realpath(q)
            if rp in seen or under_data(q) or not os.path.isfile(q) or (exclude and exclude(q)):
                continue
            seen.add(rp)
            out.append(q)
    return out

def doc_rows(p, d):
    host = is_host(p)
    if not isinstance(d, dict):
        return "other", {}, []
    if isinstance(d.get("results"), list):
        hits = [r for r in d["results"] if isinstance(r, dict) and (TICK_RE.search(jl(r)) or (not host and r.get("match_id") in IDS))]
        rec = d.get("record") or {}

                "record.vs_close": rec.get("vs_close"), "record.redated": rec.get("redated")}
        return "results", head, [results_extract(r) for r in hits]
    if d.get("kind") == "desk_parlays_v1":
        out = []
        for i, t in enumerate(d.get("tickets") or [], 1):
            for leg in t.get("legs") or []:

                if by_tick or by_team:

                                "edge_pp": t.get("edge_pp"), "matched_by": "ticker" if by_tick else "teams + kickoff (no crossed ticker on its order line)", "leg": leg})
        return "parlays", {"as_of": g(d, "desk_meta", "as_of"), "files": d.get("files")}, out
    rows = d.get("predictions") or d.get("fixtures")
    if isinstance(rows, list):
        head = {"exported_at": d.get("exported_at"), "desk_as_of": g(d, "desk_meta", "as_of"),
                "kalshi_leg_guard": g(d, "desk_meta", "kalshi_leg_guard"), "closing": d.get("closing")}
        out = []
        for r in rows:
            if isinstance(r, dict) and (TICK_RE.search(jl(r)) or (not host and r.get("match_id") in IDS)):
                x = row_extract(r)
                if host:
                    x["host_ids"] = "host DB match id: a different numbering"
                out.append(x)
        return "rows", head, out
    return "other", {}, []

def scan(tag, title, patterns, w, exclude=None):
    tag = wtag(tag, w)
    hdr(tag, wtitle(title, w))
    fs = listing(patterns, exclude)
    dts = {p: dated(p) for p in fs}
    new = [p for p in fs if inwin(dts[p][0], w)]
    nout = len(fs) - len(new)
    nfiles = nrows = bad = same = 0
    for p in new:
        try:
            b = rd(p)
        except OSError as e:
            print("UNREADABLE", p, e)
            bad += 1
            continue
        h = hashlib.sha256(b).hexdigest()
        mt, how = dts[p]
        when = ("| " + how) if how else ("| mtime " + utc(mt))
        if h in SEEN and SEEN[h] != p:
            if TICK_RB.search(b) or WIDE_RB.search(b):
                print("=", p, when, "| same bytes as", SEEN[h], "(rows not repeated)" + blab(mt))
                same += 1
            continue
        SEEN.setdefault(h, p)
        try:
            d = json.loads(b)
        except ValueError as e:
            print("UNREADABLE (not JSON)", p, e)
            bad += 1
            continue
        kind, head, rows = doc_rows(p, d)
        if not rows:
            continue
        nfiles += 1
        nrows += len(rows)
        print("FILE", p, when, "| sha256", h[:12], "|", kind, "|", jl({k: v for k, v in head.items() if v is not None}) + blab(mt))
        for r in rows:
            print("  ROW", jl(r))
    cnt(tag, "%d rows in %d files (%d files in the window read, %d outside it not read, %d same-bytes copies, %d unreadable; %d dated by a commit, not mtime; "
        "%d dated by the document\x27s own time after a squash; %d undatable (squashed) in this window, listed in every pass starting at or before its squash)"
        % (nrows, nfiles, len(new), nout, same, bad, sum(1 for p in fs if dts[p][1] and dts[p][1].startswith(("dated by its last commit", "dated by its first commit"))),
           sum(1 for p in fs if dts[p][1] and dts[p][1].startswith("dated by the document")), sum(1 for p in new if isinstance(dts[p][0], UB))))

def latest_of(o, fields, naive_utc=True):
    ep, nr = [], []
    for f in fields:
        v = o.get(f) if isinstance(o, dict) else None
        if v is None:
            continue
        e, raw, note = iso_epoch(v, naive_utc)
        if e is not None:
            ep.append((e, f, raw))
        elif note.startswith("no reliable"):
            nr.append(f + " " + str(raw))
    tail = (" | no reliable time (no Z or offset in the text, not read as UTC): " + ", ".join(nr)) if nr else ""
    if not ep:
        return None, tail
    top = max(e for e, _, _ in ep)
    return top, " | dated by " + ", ".join(f + " (raw " + raw + ")" for e, f, raw in ep if e == top) + " = " + utc(top) + blab(top) + tail
